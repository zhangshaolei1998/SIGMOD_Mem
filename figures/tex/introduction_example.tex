\begin{figure}[t]
\centering
\begin{tikzpicture}[
 font=\sffamily\footnotesize,
 box/.style={draw=#1,line width=0.8pt,rounded corners=2pt,
  fill=#1!7,text width=6.35cm,align=left,inner sep=5pt},
 flow/.style={->,>=Stealth,line width=1.0pt},
 feedback/.style={->,>=Stealth,line width=1.2pt,dashed,
  rounded corners=7pt}
]
\node[box=darkgray_x11] (raw) {
 \textbf{Raw $\Hh$: User--Agent conversation history}\\
 Mar.\ 2: \emph{Ava arranged our 4-hour outdoor hike.}\\
 Mar.\ 9: \emph{Ben booked my 2-hour indoor pottery class.}\\
 Mar.\ 16: \emph{Cara arranged our 3-hour outdoor ride.}};
\node[box=softblue,below=0.28cm of raw] (facts) {
 \textbf{Question $Q_1$ (Mar.\ 17):} \emph{Whom should I contact to repeat my
 March outdoor activities?}\\[3pt]
 \textbf{First-order $\Ff$ for query $q_1$:} settings and organizers\\
 Hike: outdoors, Ava;\quad Pottery: indoors, Ben;\\
 Ride: outdoors, Cara.\quad Durations are not yet needed.};
\node[box=cadmiumgreen,below=0.28cm of facts] (use) {
 \textbf{Second-order $\Uu$: Agent--Memory experience}\\
 \textbf{Answer $A_1$: Ava for the hike; Cara for the ride.}\\[3pt]
 Complete use report: credit all settings and Ava/Cara links.
 Ben's link was returned but unused; retain query execution cost.
 These observations change later priorities and cost estimates.};
\node[box=softorange,below=0.42cm of use] (next) {
 \textbf{Question $Q_2$ (Mar.\ 18):} \emph{Who organized my longest outdoor
 activity in March?}\\[3pt]
 \textcolor{softblue}{\textbf{Performance ($q_2$):}} verify March's
 outdoor scope; acquire durations
 $4>3$; bind the winning hike to Ava.\\[3pt]
 \textcolor{softorange}{\textbf{Efficiency:}} reuse known settings
 and organizers; budget work for the missing durations,
 not repeated judgments of every known fact.\\[3pt]
 \textbf{Answer $A_2$: Ava.} Durations enrich $\Ff$; their reported use
 and acquisition cost update $\Uu$ for future queries.};
\draw[flow,softblue] (raw.south) -- (facts.north);
\draw[flow,cadmiumgreen] (facts.south) -- (use.north);
\draw[flow,cadmiumgreen] (use.south) --
 node[midway,right,font=\scriptsize,text=cadmiumgreen,
  fill=white,inner sep=1pt] {Guide next query} (next.north);
% The two interfaces yield different persistent histories. Completed
% User--Agent sessions append to raw history; Agent--Memory operations
% update use and cost experience after each execution.
\draw[feedback,darkgray_x11] (next.west) -- ++(-0.42cm,0)
 |- (raw.west);
\node[rotate=90,font=\scriptsize\bfseries,text=darkgray_x11]
 at ($(raw.west)!0.5!(next.west)+(-0.66cm,0)$) {Append closed Q/A session};
\draw[feedback,cadmiumgreen] (next.east) -- ++(0.42cm,0)
 |- (use.east);
\node[rotate=90,font=\scriptsize\bfseries,text=cadmiumgreen]
 at ($(use.east)!0.5!(next.east)+(0.66cm,0)$) {New use + cost};
\end{tikzpicture}
\caption{Two histories evolve together. The gray loop appends completed
User--Agent conversations to raw sessions. Query-driven facts enrich
first-order memory. The green loop retains Agent--Memory use and cost
experience in second-order memory to guide later queries.
Illustrative example, not an evaluation trace.}
\Description{Alice describes a hike organized by Ava, an indoor pottery
class organized by Ben, and a ride organized by Cara. An earlier contact
question acquires their settings and organizers, but not durations.
The agent uses the settings and Ava/Cara links; Ben's link is unused.
A later longest-outdoor-activity question acquires durations, selects the hike,
and returns Ava, while use experience guides priorities and budgets.
A gray feedback loop appends each closed User--Agent conversation,
including the question and answer, to raw sessions. A green feedback
loop returns the later query's validated reported use and cost to
second-order memory, which then guides subsequent queries.}
\label{fig:motivating}
\end{figure}
